# Retry with exponential backoff + full jitter for the Claude streaming
# call. Anthropic SDK's built-in `max_retries` covers the INITIAL request
# but does NOT cover mid-stream connection drops (httpx.RemoteProtocolError
# "incomplete chunked read") — exactly the failure mode that killed the
# 5/27 kr-open slot. tenacity wraps the entire stream attempt so a dropped
# connection triggers a full restart of the streaming call.
tenacity>=8.2.0
# Robust JSON repair — Claude occasionally emits structurally broken JSON
# (mismatched brackets, missing commas) that ast.literal_eval can't handle.
json-repair>=0.30.0
# Direct market data fetch — replaces Claude's training-data hallucination
# with verified Yahoo Finance API values. Indices, stocks, FX, commodities,
# crypto all hard-fetched and injected into the prompt as ground truth.
yfinance>=0.2.40
# Exchange trading calendars — authoritative source for NYSE (XNYS),
# Korea Exchange (XKRX), and Shanghai Stock Exchange (XSHG) market
# holidays. Used to short-circuit the pipeline before any Claude/yfinance
# call on weekends and market holidays. Without this, holiday posts
# (US Independence Day, Chuseok, Chinese New Year, National Day Golden
# Week, etc.) get generated from training-data hallucination and ruin
# trust.
#
# Version pin >=4.10.0 because older versions (e.g. 4.5) ship XSHG
# calendar data only through 2025-12-31, which would silently fail the
# cn-open / cn-close holiday gate in 2026. 4.10+ extends the XSHG
# session table through 2026+.
exchange-calendars>=4.10.0
